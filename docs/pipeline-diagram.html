    F["FETCH agent · sonnet<br/>read cc-input.json, adapter context + threads, .cc-review.yaml"]
    F --> GJ["JS: normalize repo/pr; placeholder repo to empty<br/>targeted = expectedPr or post; localDryRun otherwise"]
    GJ --> G1{"diff non-empty?"}
    G1 -->|"no"| AB1["return ok:false — empty diff"]
    G1 -->|"yes"| G2{"targeted AND pr resolved AND repo set?"}
    G2 -->|"no, targeted"| AB2["abort — refuse to review a fallback diff"]
    G2 -->|"ok"| PJ["JS: changedFiles to applyIgnorePaths to mergeLangs<br/>buildProjectContext from .cc-review.yaml"]
    PJ --> T["TRIAGE agent · opus medium<br/>category, difficulty 1-5, intent.mode, overview_markdown"]
    T --> A{"intent.mode = answer?"}
    A -->|"yes"| ANS["POST-ANSWER agent · sonnet — then stop"]
    A -->|"no"| OV{"post AND mode in full/focused?"}
    OV -->|"yes"| OVP["POST-OVERVIEW agent · sonnet"]
    OV -->|"no"| PA
    OVP --> PA["JS: difficultyPath(N) picks PATH row<br/>recon? grade? verify? + opusCount + sonnetCount"]
    PA --> RV{"reviewing? not recheck/skip_new"}
    RV -->|"no"| CMG
    RV -->|"yes"| RG{"PATH.recon? difficulty >= 2"}
    RG -->|"yes"| RC["RECON agents · parallel<br/>2 sonnet scanners + 1 haiku tool-runner"]
    RG -->|"no"| SQ
    RC --> RCJ["JS: collect observations + tool findings"]
    RCJ --> SQ["JS: pickSquad by relevanceScores(obs, files, category)<br/>top opusCount domains to opus xhigh; next sonnetCount to sonnet<br/>LOW_STAKES_CATEGORY docs/chore: opus slots run on sonnet"]
    SQ --> RVW["REVIEW agents · parallel<br/>each domain gets partitioned obs + lang-refs + tools + overview"]
    RVW --> DDJ["JS: dedup findings by file:line"]
    DDJ --> HF{"findings > 0?"}
    HF -->|"no"| CMG
    HF -->|"yes"| GR{"PATH.grade? difficulty >= 2"}
    GR -->|"yes"| GRD["GRADE agent · opus high<br/>confirmed / false-positive / reframe + call_stack"]
    GR -->|"no"| GRJ["JS: trust reviewers, map all to confirmed"]
    GRD --> SPJ["JS: split false-positives vs actionable"]
    GRJ --> SPJ
    SPJ --> VR{"PATH.verify AND actionable > 0? difficulty 4-5"}
    VR -->|"yes"| VFY["VERIFY agents · parallel sonnet skeptic per finding<br/>keep vs refute"]
    VR -->|"no"| CMG
    VFY --> VJ["JS: kept vs refuted"]
    VJ --> CMG{"prior open threads > 0?"}
    CMG -->|"yes"| CM["COMMENT-MGMT agents · parallel, one per thread · sonnet<br/>validate code first, then resolve / reply / keep"]
    CMG -->|"no"| DRJ
    CM --> CMJ["JS: resolveIds, replies + comment_db_id, knownPatterns, stillOpen"]
    CMJ --> DRJ["JS: dropAlreadyThreaded — no cross-run double-post"]
    DRJ --> DEJ["JS: event = kept==0 AND stillOpen==0 ? APPROVE : REQUEST_CHANGES<br/>buildPayload + stampHash"]
    DEJ --> PG{"post = true?"}
    PG -->|"yes"| PST["POST-REVIEW agent · sonnet<br/>reply, then resolve, then post-review via adapter"]
    PG -->|"no"| MG
    PST --> MG{"confirmed or resolved or replied?"}
    MG -->|"yes"| MEM["MEMORY agent · sonnet — append to .cc-memory"]
    MG -->|"no"| END["return result"]
    MEM --> END
